CREATE SCHEMA IF NOT EXISTS devops4;
SELECT * FROM pg_catalog.pg_namespace ORDER BY nspname;
CREATE TABLE public.users (    id SERIAL PRIMARY KEY,    username VARCHAR(50) NOT NULL,    email VARCHAR(100) UNIQUE NOT NULL,    created_at TIMESTAMP DEFAULT now());
CREATE TABLE public.users (    id SERIAL PRIMARY KEY,    username VARCHAR(50) NOT NULL,    email VARCHAR(100) UNIQUE NOT NULL,    created_at TIMESTAMP DEFAULT now());
CREATE TABLE public.projects (    id SERIAL PRIMARY KEY,    name VARCHAR(100) NOT NULL,    owner_id INT REFERENCES public.users(id),    created_at TIMESTAMP DEFAULT now());
CREATE TABLE devops4.servers (    id SERIAL PRIMARY KEY,    hostname VARCHAR(100) NOT NULL,    ip_address INET NOT NULL,    created_at TIMESTAMP DEFAULT now());
CREATE TABLE devops4.services (    id SERIAL PRIMARY KEY,    name VARCHAR(100) NOT NULL,    server_id INT REFERENCES devops4.servers(id),    status VARCHAR(20) CHECK (status IN ('running','stopped','failed')),    created_at TIMESTAMP DEFAULT now());
CREATE TABLE devops4.deployments (    id SERIAL PRIMARY KEY,    service_id INT REFERENCES devops4.services(id),    version VARCHAR(50) NOT NULL,    deployed_at TIMESTAMP DEFAULT now());
